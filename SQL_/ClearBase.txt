DECLARE
 CURSOR user_cursor
  IS
  SELECT obj name
         FROM sba_objinf
         where obj <> 'SMART' and type='U'; 
 l_name   user_cursor%ROWTYPE;
BEGIN
  OPEN user_cursor;
  LOOP
    FETCH user_cursor INTO l_name;
    EXIT WHEN user_cursor%NOTFOUND;    
    execute immediate 'delete from sba_objinf where OBJ = '''||l_name.name||'''';
    execute immediate 'drop user '||l_name.name||' cascade'; 
  END LOOP;
  CLOSE user_cursor;
  commit;
END;
